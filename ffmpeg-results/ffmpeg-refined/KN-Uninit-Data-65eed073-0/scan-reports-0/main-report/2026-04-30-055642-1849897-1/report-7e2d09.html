405 if (!avctx->nb_streams)
406 return AVERROR_EOF(-(int)(('E') | (('O') << 8) | (('F') << 16) | ((
unsigned)(' ') << 24)));
407
408 st = avctx->streams[mlv->stream_index];
409 if (mlv->pts >= st->duration)
410 return AVERROR_EOF(-(int)(('E') | (('O') << 8) | (('F') << 16) | ((
unsigned)(' ') << 24)));
411
412 index = av_index_search_timestamp(st, mlv->pts, AVSEEK_FLAG_ANY4);
413 if (index < 0) {
414 av_log(avctx, AV_LOG_ERROR16, "could not find index entry for frame %"PRId64"l" "d""\n", mlv->pts);
415 return AVERROR(EIO)(-(5));
416 }
417
418 pb = mlv->pb[st->internal->index_entries[index].size];
419 if (!pb) {
420 ret = FFERROR_REDO(-(int)(('R') | (('E') << 8) | (('D') << 16) | ((
unsigned)('O') << 24)));
421 goto next_packet;
422 }
423 avio_seek(pb, st->internal->index_entries[index].pos, SEEK_SET0);
424
425 avio_skip(pb, 4); // blockType
426 size = avio_rl32(pb);
427 if (size < 16)
428 return AVERROR_INVALIDDATA(-(int)(('I') | (('N') << 8) | (('D') << 16) | ((
unsigned)('A') << 24)));
429 avio_skip(pb, 12); //timestamp, frameNumber
430 if (st->codecpar->codec_type == AVMEDIA_TYPE_VIDEO)
431 avio_skip(pb, 8); // cropPosX, cropPosY, panPosX, panPosY
432 space = avio_rl32(pb);
433 avio_skip(pb, space);
434
435 if ((mlv->class[st->id] & (MLV_CLASS_FLAG_DELTA0x40|MLV_CLASS_FLAG_LZMA0x80))) {
436 ret = AVERROR_PATCHWELCOME(-(int)(('P') | (('A') << 8) | (('W') << 16) | ((
unsigned)('E') << 24)));
437 } else if (st->codecpar->codec_type == AVMEDIA_TYPE_VIDEO) {
438 ret = av_get_packet(pb, pkt, (st->codecpar->width * st->codecpar->height * st->codecpar->bits_per_coded_sample + 7) >> 3);
439 } else { // AVMEDIA_TYPE_AUDIO
440 if (space > UINT_MAX(2147483647 *2U +1U) - 24 || size < (24 + space))
441 return AVERROR_INVALIDDATA(-(int)(('I') | (('N') << 8) | (('D') << 16) | ((
unsigned)('A') << 24)));
442 ret = av_get_packet(pb, pkt, size - (24 + space));
443 }
444
445 if (ret < 0)
446 return ret;
447
448 pkt->stream_index = mlv->stream_index;
449 pkt->pts = mlv->pts;
450
451 ret = 0;
452next_packet:
453 mlv->stream_index++;
454 if (mlv->stream_index == avctx->nb_streams) {
455 mlv->stream_index = 0;
456 mlv->pts++;
457 }
458 return ret;
459}

461static int read_seek(AVFormatContext *avctx, int stream_index, int64_t timestamp, int flags)
462{
463 MlvContext *mlv = avctx->priv_data;
464
465 if ((flags & AVSEEK_FLAG_FRAME8) || (flags & AVSEEK_FLAG_BYTE2))
466 return AVERROR(ENOSYS)(-(38));
467
468 if (!(avctx->pb->seekable & AVIO_SEEKABLE_NORMAL(1 << 0)))
469 return AVERROR(EIO)(-(5));
470
471 mlv->pts = timestamp;
472 return 0;
473}

475static int read_close(AVFormatContext *s)
476{
477 MlvContext *mlv = s->priv_data;
478 int i;
479 for (i = 0; i < 100; i++)
480 ff_format_io_close(s, &mlv->pb[i]);
481 return 0;
482}
