410 AV_WL32(pkt->data , nblocks)((((union unaligned_32 *) (pkt->data))->l) = (nblocks));
411 AV_WL32(pkt->data + 4, ape->frames[ape->currentframe].skip)((((union unaligned_32 *) (pkt->data + 4))->l) = (ape->

412 ret = avio_read(s->pb, pkt->data + extra_size, ape->frames[ape->currentframe].size);
413 if (ret < 0) {

415 }
416
417 pkt->pts = ape->frames[ape->currentframe].pts;
418 pkt->stream_index = 0;

420 /* note: we need to modify the packet size here to handle the last

422 pkt->size = ret + extra_size;
