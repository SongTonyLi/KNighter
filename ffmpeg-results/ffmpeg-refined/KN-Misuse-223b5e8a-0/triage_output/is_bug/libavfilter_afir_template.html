127 s->fdsp->vector_fmac_scalar(dst, src, 1.f, nb_samples);
128#else
129 s->fdsp->vector_dmac_scalar(dst, src, 1.0, nb_samples);
130#endif
131 } else {
132 for (int n = 0; n < nb_samples; n++)
133 dst[n] += src[n];
134 }
135}

137static int fn(fir_quantum)fir_quantum_double(AVFilterContext *ctx, AVFrame *out, int ch, int ioffset, int offset, int selir)
138{
139 AudioFIRContext *s = ctx->priv;
140 const ftypedouble *in = (const ftypedouble *)s->in->extended_data[ch] + ioffset;
141 ftypedouble *blockout, *ptr = (ftypedouble *)out->extended_data[ch] + offset;
142 const int min_part_size = s->min_part_size;
143 const int nb_samples = FFMIN(min_part_size, out->nb_samples - offset)((min_part_size) > (out->nb_samples - offset) ? (out->
nb_samples - offset) : (min_part_size));
144 const int nb_segments = s->nb_segments[selir];
145 const float dry_gain = s->dry_gain;
146 const float wet_gain = s->wet_gain;
147
148 for (int segment = 0; segment < nb_segments; segment++) {
149 AudioFIRSegment *seg = &s->seg[selir][segment];
150 ftypedouble *src = (ftypedouble *)seg->input->extended_data[ch];
151 ftypedouble *dst = (ftypedouble *)seg->output->extended_data[ch];
152 ftypedouble *sumin = (ftypedouble *)seg->sumin->extended_data[ch];
153 ftypedouble *sumout = (ftypedouble *)seg->sumout->extended_data[ch];
154 ftypedouble *tempin = (ftypedouble *)seg->tempin->extended_data[ch];
155 ftypedouble *buf = (ftypedouble *)seg->buffer->extended_data[ch];
156 int *output_offset = &seg->output_offset[ch];
157 const int nb_partitions = seg->nb_partitions;
158 const int input_offset = seg->input_offset;
159 const int part_size = seg->part_size;
160 int j;
161
162 seg->part_index[ch] = seg->part_index[ch] % nb_partitions;
163 if (dry_gain == 1.f) {
164 memcpy(src + input_offset, in, nb_samples * sizeof(*src));
165 } else if (min_part_size >= 8) {
166#if DEPTH64 == 32
167 s->fdsp->vector_fmul_scalar(src + input_offset, in, dry_gain, FFALIGN(nb_samples, 4)(((nb_samples)+(4)-1)&~((4)-1)));
168#else
169 s->fdsp->vector_dmul_scalar(src + input_offset, in, dry_gain, FFALIGN(nb_samples, 8)(((nb_samples)+(8)-1)&~((8)-1)));
170#endif
171 } else {
172 ftypedouble *src2 = src + input_offset;
173 for (int n = 0; n < nb_samples; n++)
174 src2[n] = in[n] * dry_gain;
175 }
176
177 output_offset[0] += min_part_size;
178 if (output_offset[0] >= part_size) {
179 output_offset[0] = 0;
180 } else {
181 memmove(src, src + min_part_size, (seg->input_size - min_part_size) * sizeof(*src));
182
183 dst += output_offset[0];
184 fn(fir_fadd)fir_fadd_double(s, ptr, dst, nb_samples);
185 continue;
186 }
187
188 memset(sumin, 0, sizeof(*sumin) * seg->fft_length);
189
190 blockout = (ftypedouble *)seg->blockout->extended_data[ch] + seg->part_index[ch] * seg->block_size;
191 memset(tempin + part_size, 0, sizeof(*tempin) * (seg->block_size - part_size));
192 memcpy(tempin, src, sizeof(*src) * part_size);
193 seg->tx_fn(seg->tx[ch], blockout, tempin, sizeof(ftypedouble));
194
195 j = seg->part_index[ch];
196 for (int i = 0; i < nb_partitions; i++) {
197 const int input_partition = j;
198 const int coeff_partition = i;
199 const int coffset = coeff_partition * seg->coeff_size;
200 const ftypedouble *blockout = (const ftypedouble *)seg->blockout->extended_data[ch] + input_partition * seg->block_size;
201 const ctypeAVComplexDouble *coeff = ((const ctypeAVComplexDouble *)seg->coeff->extended_data[ch]) + coffset;
202
203 if (j == 0)
204 j = nb_partitions;
205 j--;
206
207#if DEPTH64 == 32
208 s->afirdsp.fcmul_add(sumin, blockout, (const ftypedouble *)coeff, part_size);
209#else
210 s->afirdsp.dcmul_add(sumin, blockout, (const ftypedouble *)coeff, part_size);
211#endif
212 }
213
214 seg->itx_fn(seg->itx[ch], sumout, sumin, sizeof(ctypeAVComplexDouble));
215
216 fn(fir_fadd)fir_fadd_double(s, buf, sumout, part_size);
217 memcpy(dst, buf, part_size * sizeof(*dst));
218 memcpy(buf, sumout + part_size, part_size * sizeof(*buf));
219
220 fn(fir_fadd)fir_fadd_double(s, ptr, dst, nb_samples);
221
222 if (part_size != min_part_size)
223 memmove(src, src + min_part_size, (seg->input_size - min_part_size) * sizeof(*src));
224
225 seg->part_index[ch] = (seg->part_index[ch] + 1) % nb_partitions;
226 }
227
228 if (wet_gain == 1.f)
229 return 0;
230
231 if (min_part_size >= 8) {
232#if DEPTH64 == 32
233 s->fdsp->vector_fmul_scalar(ptr, ptr, wet_gain, FFALIGN(nb_samples, 4)(((nb_samples)+(4)-1)&~((4)-1)));
234#else
235 s->fdsp->vector_dmul_scalar(ptr, ptr, wet_gain, FFALIGN(nb_samples, 8)(((nb_samples)+(8)-1)&~((8)-1)));
236#endif
237 } else {
238 for (int n = 0; n < nb_samples; n++)
239 ptr[n] *= wet_gain;
240 }
241
242 return 0;
243}

245static void fn(fir_quantums)fir_quantums_double(AVFilterContext *ctx, AudioFIRContext *s, AVFrame *out,
246 int min_part_size, int ch, int offset,
247 int prev_selir, int selir)
248{
249 if (ctx->is_disabled || s->prev_is_disabled) {
250 const ftypedouble *in = (const ftypedouble *)s->in->extended_data[ch] + offset;
251 const ftypedouble *xfade0 = (const ftypedouble *)s->xfade[0]->extended_data[ch];
252 const ftypedouble *xfade1 = (const ftypedouble *)s->xfade[1]->extended_data[ch];
253 ftypedouble *src0 = (ftypedouble *)s->fadein[0]->extended_data[ch];
254 ftypedouble *src1 = (ftypedouble *)s->fadein[1]->extended_data[ch];
255 ftypedouble *dst = ((ftypedouble *)out->extended_data[ch]) + offset;
256
257 if (ctx->is_disabled && !s->prev_is_disabled) {
258 memset(src0, 0, min_part_size * sizeof(ftypedouble));
259 fn(fir_quantum)fir_quantum_double(ctx, s->fadein[0], ch, offset, 0, selir);
260 for (int n = 0; n < min_part_size; n++)
261 dst[n] = xfade1[n] * src0[n] + xfade0[n] * in[n];
262 } else if (!ctx->is_disabled && s->prev_is_disabled) {
263 memset(src1, 0, min_part_size * sizeof(ftypedouble));
264 fn(fir_quantum)fir_quantum_double(ctx, s->fadein[1], ch, offset, 0, selir);
265 for (int n = 0; n < min_part_size; n++)
266 dst[n] = xfade1[n] * in[n] + xfade0[n] * src1[n];
267 } else {
268 memcpy(dst, in, sizeof(ftypedouble) * min_part_size);
269 }
270 } else if (prev_selir != selir && s->loading[ch] != 0) {
271 const ftypedouble *xfade0 = (const ftypedouble *)s->xfade[0]->extended_data[ch];
272 const ftypedouble *xfade1 = (const ftypedouble *)s->xfade[1]->extended_data[ch];
273 ftypedouble *src0 = (ftypedouble *)s->fadein[0]->extended_data[ch];
274 ftypedouble *src1 = (ftypedouble *)s->fadein[1]->extended_data[ch];
275 ftypedouble *dst = ((ftypedouble *)out->extended_data[ch]) + offset;
276
277 memset(src0, 0, min_part_size * sizeof(ftypedouble));
278 memset(src1, 0, min_part_size * sizeof(ftypedouble));
279
280 fn(fir_quantum)fir_quantum_double(ctx, s->fadein[0], ch, offset, 0, prev_selir);
281 fn(fir_quantum)fir_quantum_double(ctx, s->fadein[1], ch, offset, 0, selir);
282
283 if (s->loading[ch] > s->max_offset[selir]) {
284 for (int n = 0; n < min_part_size; n++)
285 dst[n] = xfade1[n] * src0[n] + xfade0[n] * src1[n];
286 s->loading[ch] = 0;
287 } else {
288 memcpy(dst, src0, min_part_size * sizeof(ftypedouble));
289 }
290 } else {
291 fn(fir_quantum)fir_quantum_double(ctx, out, ch, offset, offset, selir);
292 }
